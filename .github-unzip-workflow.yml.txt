# Optional: copy this into .github/workflows/unzip.yml on GitHub (see README_DEPLOY.md)
name: Unzip handoverlab
on: workflow_dispatch
permissions:
  contents: write
jobs:
  unzip:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - run: |
          unzip -q handoverlab.zip
          cp -r handoverlab/. .
          rm -rf handoverlab handoverlab.zip
      - run: |
          git config user.name "github-actions"
          git config user.email "actions@github.com"
          git add -A
          git commit -m "Unzip HandoverLab project"
          git push
